#include &lt;fstream&gt;

    std::ifstream inputFile("example.txt"); // Открытие файла для чтения

    if (!inputFile.is_open()) {
        std::cerr &lt;&lt; "Не удалось открыть файл." &lt;&lt; std::endl;

    }

    inputFile.close(); // Закрытие файла

    return 0;
}
